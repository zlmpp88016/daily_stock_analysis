```typescript
export const agentApi = {
  chat: async (message: string, sessionId?: string) => {
    const response = await fetch('/api/v1/agent/chat', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ message, session_id: sessionId })
    });
    return response.json();
  },

  clearSession: async (sessionId: string) => {

      method: 'DELETE'
    });
  }
};
```

### 3.2 命令系统

#### /chat 命令 (bot/commands/chat.py)

```python
class ChatCommand(Command):
    """Agent 对话命令"""

    def execute(self, message: BotMessage, args: list[str]) -> BotResponse:
        # 1. 检查 Agent 模式是否开启
        if not config.agent_mode:
            return BotResponse.text_response(
                "⚠️ Agent 模式未开启，无法使用对话功能。"
            )

        # 2. 解析用户消息
        user_message = " ".join(args)
        session_id = f"{message.platform}_{message.user_id}"

        # 3. 调用 Agent 执行器
        from src.agent.factory import build_agent_executor
        executor = build_agent_executor(config)
        result = executor.chat(message=user_message, session_id=session_id)

        # 4. 返回响应
        if result.success:
            return BotResponse.text_response(result.content)
        else:
            return BotResponse.text_response(f"⚠️ 对话失败: {result.error}")
```

#### /analyze 命令 (bot/commands/analyze.py)

```python
class AnalyzeCommand(Command):
    """股票分析命令"""

    def execute(self, message: BotMessage, args: list[str]) -> BotResponse:
        # 1. 解析股票代码
        stock_codes = self._parse_codes(args)

        # 2. 创建分析流水线
        pipeline = StockAnalysisPipeline(
            source_message=message,
            query_id=uuid.uuid4().hex,
            query_source="bot"
        )

        # 3. 批量分析
        results = []
        for code in stock_codes:
            result = pipeline.process_single_stock(code)
            if result:
                results.append(result)

        # 4. 格式化响应
        return self._format_results(results)
```

### 3.3 消息流程

```
用户在 Bot 中发送命令
    ↓
Bot 平台接收消息
    ↓
Webhook 转发到系统
    ↓
CommandRouter 解析命令
    ↓
执行对应 Command
    ↓
调用业务层 (Pipeline/Agent)
    ↓
生成响应
    ↓
通过 NotificationService 推送
    ↓
用户收到回复
```

```python
@router.post("/chat")
async def agent_chat(request: AgentChatRequest):
    """Agent 对话接口"""

    # 1. 构建 Agent 执行器
    config = get_config()
    executor = build_agent_executor(config)

    # 2. 执行对话
    result = executor.chat(
        message=request.message,
        session_id=request.session_id
    )

    # 3. 返回结果
    return {
        "success": result.success,
        "content": result.content,
        "dashboard": result.dashboard,
        "session_id": request.session_id,
        "total_steps": result.total_steps,
        "total_tokens": result.total_tokens,

    }
```

## 五、数据流详解

### 5.1 传统分析流程

```
API/Bot/CLI 请求
    ↓
StockAnalysisPipeline.process_single_stock()
    ↓
├─ fetch_and_save_stock_data()
│   └─ DataFetcherManager.get_daily_data()
    ↓
├─ analyze_stock()
│   ├─ DataFetcherManager.get_realtime_quote()
│   ├─ DataFetcherManager.get_chip_distribution()
│   ├─ StockTrendAnalyzer.analyze()
│   ├─ SearchService.search_comprehensive_intel()
│   ├─ Database.get_analysis_context()
│   └─ GeminiAnalyzer.analyze()
    ↓
├─ Database.save_analysis_history()
    ↓
└─ NotificationService.send_analysis_result()
```

### 5.2 Agent 对话流程

```
Web/Bot/API 请求
    ↓
AgentExecutor.chat()
    ↓
├─ ConversationManager.get_or_create(session_id)
├─ 加载会话历史
    ↓
[ReAct 循环开始]
    ↓
├─ LLMToolAdapter.call_with_tools()
│   ├─ 构建系统提示词
│   ├─ 添加工具声明
│   └─ 发送到 LLM
    ↓
├─ 解析响应
│   ├─ 工具调用？
│   │   ├─ ToolRegistry.execute_tool()
│   │   ├─ 记录工具调用日志
│   │   └─ 返回工具结果
│   └─ 文本响应？
│       └─ 解析为最终答案
    ↓
├─ 达到最大步数或获得最终答案？
│   ├─ NO → 继续循环
│   └─ YES → 退出循环
    ↓
├─ ConversationManager.add_message()
├─ Database.save_conversation_message()
    ↓
└─ 返回 AgentResult
```

## 六、通知推送

### 6.1 NotificationService

**支持渠道**:
- 企业微信 (WeCom)
- 飞书 (Lark)
- Telegram
- 钉钉 (DingTalk)
- 邮件 (Email)
- Pushover

**推送时机**:
1. 定时分析完成后
2. Bot 命令触发分析后
3. API 请求分析后（可选）

**消息格式**:

```python
def send_analysis_result(self, result: AnalysisResult):
    """发送分析结果"""

    # 1. 格式化消息
    message = self._format_analysis_message(result)

    # 2. 发送到各个渠道
    for channel in self.enabled_channels:
        try:
            if channel == NotificationChannel.WECOM:
                self._send_wecom(message)
            elif channel == NotificationChannel.LARK:
                self._send_lark(message)
            elif channel == NotificationChannel.TELEGRAM:
                self._send_telegram(message)
            # ... 其他渠道
        except Exception as e:
            logger.error(f"发送到 {channel} 失败: {e}")
```

## 八、错误处理

### 8.1 API 错误响应

```python
@app.exception_handler(Exception)
async def global_exception_handler(request: Request, exc: Exception):
    """全局异常处理"""
    logger.error(f"Unhandled exception: {exc}", exc_info=True)

    return JSONResponse(
        status_code=500,
        content={
            "success": false,
            "error": str(exc),
            "detail": "Internal server error"
        }
    )
```

### 8.2 Bot 错误处理

```python
try:
    result = executor.chat(message=user_message, session_id=session_id)
```

## 九、性能优化

### 9.1 并发控制

```python
# 批量分析时使用线程池
with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
    futures = {
        executor.submit(self.process_single_stock, code): code
        for code in stock_codes
    }

    for future in as_completed(futures):
        code = futures[future]
        try:
            result = future.result()
            results.append(result)
        except Exception as e:
            logger.error(f"分析 {code} 失败: {e}")
```

## 十二、总结

### 12.1 集成方式对比

| 方式 | 优势 | 适用场景 |
|------|------|----------|
| **Web** | 交互体验好，可视化强 | 个人用户、实时对话 |
| **Bot** | 便捷，无需打开网页 | 团队协作、移动端 |
| **API** | 灵活，可集成到其他系统 | 自动化、第三方集成 |

### 12.2 技术亮点

1. **统一业务层**: 三种访问方式共享同一套业务逻辑
2. **ReAct 架构**: Agent 模式支持复杂的多步推理
3. **工具系统**: 可扩展的工具注册表
4. **会话管理**: 支持多轮对话的上下文记忆
5. **多渠道推送**: 灵活的通知系统
6. **数据持久化**: 完整的历史记录和回溯能力
